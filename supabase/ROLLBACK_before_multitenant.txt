ptitle text; other record; BEGIN IF NOT public.is_admin() THEN RAISE EXCEPTION 'Not allowed'; END IF; SELECT * INTO a FROM public.applications WHERE id = _app; IF a.id IS NULL OR a.status IN ('draft','withdrawn') THEN RAISE EXCEPTION 'This application cannot be awarded'; END IF; IF EXISTS (SELECT 1 FROM public.awards WHERE project_id = a.project_id) THEN RAISE EXCEPTION 'This project has already been awarded'; END IF; INSERT INTO public.awards(project_id, application_id, value, notes) VALUES (a.project_id, a.id, _value, _notes); UPDATE public.applications SET status = 'awarded' WHERE id = a.id; SELECT title INTO ptitle FROM public.projects WHERE id = a.project_id; FOR other IN SELECT id, organization_id FROM public.applications WHERE project_id = a.project_id AND id <> a.id AND status IN ('submitted','under_review','shortlisted') LOOP UPDATE public.applications SET status = 'unsuccessful' WHERE id = other.id; PERFORM public.notify_org(other.organization_id, 'status', 'Outcome: ' || ptitle, 'Your submission was not successful on this occasion.', '/portal/applications/' || other.id); END LOOP; UPDATE public.projects SET status = 'awarded', updated_at = now() WHERE id = a.project_id;
